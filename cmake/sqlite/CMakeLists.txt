cmake_minimum_required(VERSION 3.22)
project(IDOSProjSQLite LANGUAGES C)
add_library(sqlite3 STATIC "${SQLITE_SOURCE}/sqlite3.c")
target_compile_definitions(sqlite3 PRIVATE SQLITE_THREADSAFE=1)
set_target_properties(sqlite3 PROPERTIES DEBUG_POSTFIX d)
add_executable(sqlite3_tool "${SQLITE_SOURCE}/shell.c" "${SQLITE_SOURCE}/sqlite3.c")
set_target_properties(sqlite3_tool PROPERTIES OUTPUT_NAME sqlite3)
install(TARGETS sqlite3 sqlite3_tool ARCHIVE DESTINATION lib RUNTIME DESTINATION bin)
install(FILES "${SQLITE_SOURCE}/sqlite3.h" "${SQLITE_SOURCE}/sqlite3ext.h" DESTINATION include)
